cmake_minimum_required(VERSION 3.24)
project(OpenCap VERSION 0.1.0 LANGUAGES CXX)

if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "OpenCap targets 64-bit Windows.")
endif()

include(FetchContent)
include(CTest)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")

option(OPENCAP_RECEIVER "Build physical UDCAP receiver support" ON)
set(OPENVR_SDK_ROOT "" CACHE PATH "Path to an existing OpenVR SDK")
if(NOT OPENVR_SDK_ROOT)
    FetchContent_Declare(openvr
        URL https://github.com/ValveSoftware/openvr/archive/0924064316de3effbcd1acf1e309182a2deb1c05.tar.gz
        SOURCE_SUBDIR unused)
    FetchContent_MakeAvailable(openvr)
    set(OPENVR_SDK_ROOT "${openvr_SOURCE_DIR}")
endif()

add_library(openvr_client STATIC IMPORTED)
set_target_properties(openvr_client PROPERTIES
    IMPORTED_LOCATION "${OPENVR_SDK_ROOT}/lib/win64/openvr_api.lib"
    INTERFACE_INCLUDE_DIRECTORIES "${OPENVR_SDK_ROOT}/headers")

add_library(opencap_bridge STATIC src/bridge.cpp src/model.cpp src/osc_compat.cpp)
target_include_directories(opencap_bridge PUBLIC src)
target_compile_definitions(opencap_bridge PUBLIC UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(opencap_bridge PUBLIC "$<$<CXX_COMPILER_ID:MSVC>:/utf-8>")

add_library(opencap_udcap_preferences STATIC src/udcap_preferences.cpp)
target_link_libraries(opencap_udcap_preferences PUBLIC opencap_bridge PRIVATE windowsapp xmllite shlwapi ole32 ws2_32)
add_library(opencap_firmware STATIC src/firmware_image.cpp)
target_include_directories(opencap_firmware PUBLIC src)
target_link_libraries(opencap_firmware PRIVATE bcrypt)
add_executable(OpenCap WIN32 src/desktop.cpp src/desktop_main.cpp src/desktop_preferences.cpp src/desktop_view.cpp
    src/hand_view.cpp src/input_dialog.cpp src/stream.cpp)
target_link_libraries(OpenCap PRIVATE opencap_bridge opencap_udcap_preferences opencap_firmware shell32 ole32 comdlg32 ws2_32 sapi)
if(OPENCAP_RECEIVER)
    set(BUILD_TEST_TOOLS OFF CACHE BOOL "" FORCE)
    set(COPY_HIDAPI_DLL OFF CACHE BOOL "" FORCE)
    FetchContent_Declare(community_core
        URL https://github.com/OldestNova/UdCap-Community-HandDriver-Core/archive/14987ad6751761fde0f587f4687cb6aaa8e46414.tar.gz)
    FetchContent_MakeAvailable(community_core)
    target_sources(OpenCap PRIVATE src/receiver.cpp)
    target_link_libraries(OpenCap PRIVATE UdCapCommunityDriverCore)
    target_compile_definitions(OpenCap PRIVATE OPENCAP_RECEIVER=1)
endif()

set(hand_sample "${OPENVR_SDK_ROOT}/samples/drivers/drivers/handskeletonsimulation/src")
add_library(opencap_vr_skeleton STATIC src/vr_skeleton.cpp "${hand_sample}/hand_simulation.cpp")
target_include_directories(opencap_vr_skeleton PUBLIC "${OPENVR_SDK_ROOT}/headers" PRIVATE
    "${hand_sample}" "${OPENVR_SDK_ROOT}/samples/drivers/utils/vrmath")
target_link_libraries(opencap_vr_skeleton PUBLIC opencap_bridge)
add_library(driver_opencap SHARED src/driver.cpp)
target_link_libraries(driver_opencap PRIVATE opencap_vr_skeleton)

add_executable(OpenCapDashboard WIN32 src/dashboard.cpp)
target_link_libraries(OpenCapDashboard PRIVATE opencap_bridge openvr_client gdi32)

install(TARGETS OpenCap OpenCapDashboard RUNTIME DESTINATION .)
install(TARGETS driver_opencap RUNTIME DESTINATION steamvr/opencap/bin/win64)
install(DIRECTORY steamvr/opencap/ DESTINATION steamvr/opencap)
install(FILES steamvr/opencap.vrmanifest DESTINATION .)
install(FILES scripts/Set-SteamVRRegistration.ps1 scripts/Install-OpenCap.ps1 DESTINATION .)
install(FILES LICENSE DESTINATION .)
if(EXISTS "${OPENVR_SDK_ROOT}/bin/win64/openvr_api.dll")
    install(FILES "${OPENVR_SDK_ROOT}/bin/win64/openvr_api.dll" DESTINATION .)
endif()
if(EXISTS "${OPENVR_SDK_ROOT}/LICENSE")
    install(FILES "${OPENVR_SDK_ROOT}/LICENSE" DESTINATION licenses RENAME OpenVR.txt)
endif()
if(OPENCAP_RECEIVER)
    install(FILES "${community_core_SOURCE_DIR}/LICENSE" DESTINATION licenses RENAME Community-Core.txt)
    install(FILES $<TARGET_RUNTIME_DLLS:OpenCap> DESTINATION .)
endif()

if(BUILD_TESTING)
    add_executable(OpenCapChecks tests/checks.cpp)
    target_link_libraries(OpenCapChecks PRIVATE opencap_bridge)
    target_include_directories(OpenCapChecks PRIVATE "${OPENVR_SDK_ROOT}/headers")
    add_test(NAME OpenCapChecks COMMAND OpenCapChecks)
    add_test(NAME OpenCapDriverFactory COMMAND OpenCapChecks --driver "$<TARGET_FILE:driver_opencap>")
    add_test(NAME OpenCapDesktopDemo COMMAND OpenCapChecks --desktop "$<TARGET_FILE:OpenCap>")
    set_tests_properties(OpenCapDesktopDemo PROPERTIES TIMEOUT 20)
    add_executable(OpenCapUdcapChecks tests/udcap_preferences.cpp)
    target_link_libraries(OpenCapUdcapChecks PRIVATE opencap_udcap_preferences)
    add_test(NAME OpenCapUdcapChecks COMMAND OpenCapUdcapChecks)
    add_executable(OpenCapSkeletonChecks tests/vr_skeleton.cpp)
    target_link_libraries(OpenCapSkeletonChecks PRIVATE opencap_vr_skeleton)
    add_test(NAME OpenCapSkeletonChecks COMMAND OpenCapSkeletonChecks)
    add_executable(OpenCapFirmwareChecks tests/firmware_image.cpp)
    target_link_libraries(OpenCapFirmwareChecks PRIVATE opencap_firmware)
    add_test(NAME OpenCapFirmwareChecks COMMAND OpenCapFirmwareChecks)
endif()
